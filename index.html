  "Amostras": v.length,

  "Vibração (mm/s)": d3.mean(v, d => d.vibracao_rms_mm_s),

raw = FileAttachment("dados.csv").csv({typed: true})
apps = ["bomba", "compressor", "esteira", "ventilador"]
nMotores = String(new Set(dados.map(d => d.codigo_motor)).size)
cargaMedia = (d3.mean(dados, d => d.carga) * 100).toFixed(1).replace(".", ",")
pctAlerta = (d3.mean(dados, d => d.vibracao_rms_mm_s > 2.8) * 100).toFixed(1).replace(".", ",")
corIE = ["#9c6b4e", "#6cc5b0"]
dias = ["seg", "ter", "qua", "qui", "sex", "sáb", "dom"]
corApp = ["#4269d0", "#e15759", "#efb118", "#3ca951"]
passo = Math.ceil(dados.length / 1000)
amostra = dados.filter((d,i) => i % passo === 0)
faixas = d3.rollups(
    dados,
    v => ({n: v.length, media: d3.mean(v, d => d.vibracao_rms_mm_s)}),
    d => Math.floor(d.horas_desde_manutencao / 250) * 250 + 125
  )
  .map(([h, s]) => ({h, ...s}))
  .filter(f => f.n >= 25)
  .sort((a, b) => a.h - b.h)
viewof filtroApp = Inputs.checkbox(["bomba", "compressor", "esteira", "ventilador"], { label: "Aplicações", value: ["bomba", "compressor", "esteira", "ventilador"] })

viewof inicio = Inputs.date({ label: "Data inicial", value: new Date("2024-01-01"), min: new Date("2024-01-01"), max: new Date("2025-12-31") })
viewof fim = Inputs.date({ label: "Data final", value: new Date("2025-12-31"), min: new Date("2024-01-01"), max: new Date("2025-12-31") })
dados = raw.filter(d => d.data >= inicio && d.data <= fim && filtroApp.includes(d.aplicacao))
